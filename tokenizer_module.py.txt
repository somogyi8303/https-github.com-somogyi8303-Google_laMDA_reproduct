import torch
from transformers import AutoTokenizer

class LaMDATokenizer:
    """
    The linguistic interface for Somogyi-LaMDA.
    Task: Text -> Numbers (Encode) & Numbers -> Text (Decode).
    Base: xlm-roberta (Optimized for Hungarian and Python code).
    """
    def __init__(self, base_model="xlm-roberta-base"):
        print(f"Loading tokenizer: {base_model}...")
        # XLM-RoBERTa is a massive multilingual model, excellent for Hungarian
        self.tokenizer = AutoTokenizer.from_pretrained(base_model)
        self.vocab_size = self.tokenizer.vocab_size
        print(f"Tokenizer ready. Vocab size: {self.vocab_size} tokens.")

    def encode(self, text, max_length=128):
        """
        Converts text into tensors that the model can process.
        """
        encoding = self.tokenizer(
            text,
            return_tensors='pt', # Output as PyTorch tensor
            padding='max_length',
            truncation=True,
            max_length=max_length
        )
        return encoding['input_ids'], encoding['attention_mask']

    def decode(self, token_ids):
        """
        Converts the model's numerical output back into readable text.
        """
        # skip_special_tokens=True removes technical marks like <pad>, <s>
        return self.tokenizer.decode(token_ids[0], skip_special_tokens=True)
